        // State Management

        // Tab Switching Logic
        function switchTab(tabId) {
            currentTab = tabId;

            // Hide all sections
            document.getElementById('section-home').classList.add('hidden');
            document.getElementById('section-grade1').classList.add('hidden');
            document.getElementById('section-grade2').classList.add('hidden');
            document.getElementById('section-admin').classList.add('hidden');

            // Reset Nav Active classes
            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('text-cyan-400', 'bg-cyan-950/40', 'border', 'border-cyan-500/30');
                btn.classList.add('text-gray-300');
            });

            // Show selected section
            if(tabId === 'home') {
                document.getElementById('section-home').classList.remove('hidden');
                setNavActive('nav-home');
            } else if(tabId === 'grade1') {
                document.getElementById('section-grade1').classList.remove('hidden');
                setNavActive('nav-grade1');
            } else if(tabId === 'grade2') {
                document.getElementById('section-grade2').classList.remove('hidden');

                if(isAdminLoggedIn) {
                    document.getElementById('section-admin').classList.remove('hidden');
                    renderAdminOrders();
                } else {
                    openAdminModal();
                }
            }
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function setNavActive(navId) {
            const btn = document.getElementById(navId);
            if(btn) {
                btn.classList.add('text-cyan-400', 'bg-cyan-950/40', 'border', 'border-cyan-500/30');
                btn.classList.remove('text-gray-300');

        }

        // Copy Text Helper
        function copyToClipboard(text) {
            const dummy = document.createElement("input");
            document.body.appendChild(dummy);
            dummy.value = text;
            dummy.select();
            document.execCommand("copy");
            document.body.removeChild(dummy);
            showToast("تم نسخ الرقم إلى الحافظة بنجاح!");
        }

        // Image Preview Handler
        function previewImage(input, previewId) {
            const preview = document.getElementById(previewId);
            const container = document.getElementById(previewId + '-container');
            const file = input.files[0];

            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    preview.src = e.target.result;
                    preview.classList.remove('hidden');
                    if(container) container.classList.add('hidden');
                }
                reader.readAsDataURL(file);
            }
        }

        // Form Submission Logic Grade 1 & 2
        function handleOrderSubmit(event, gradeName, categoryName) {
            event.preventDefault();
            const form = event.target;
            const name = form.querySelector('input[type="text"]').value;
            const phone = form.querySelector('input[type="tel"]').value;
            const details = form.querySelector('textarea').value;
            const fileInput = form.querySelector('input[type="file"]');

            if (!fileInput.files || !fileInput.files[0]) {
                showToast("برجاء رفع صورة التحويل أولاً!");
                return;
            }

            const reader = new FileReader();
            reader.onload = function(e) {
                const base64Img = e.target.result;
                const order = {
                    id: Date.now(),
                    date: new Date().toLocaleString('ar-EG'),
                    studentName: name,
                    phone: phone,
                    grade: gradeName,
                    category: categoryName,
                    details: details,
                    receiptImg: base64Img
                };

                saveOrderToStorage(order);
                form.reset();
                showToast("تم إرسال طلبك وصورة التحويل بنجاح! سيتم التواصل معك قريباً.");
                
                // Reset Preview
                document.getElementById('preview-g1').classList.add('hidden');
                document.getElementById('preview-g1-container').classList.remove('hidden');
            };
            reader.readAsDataURL(fileInput.files[0]);
        }

        // Local Storage Handling
        function saveOrderToStorage(order) {
            let orders = JSON.parse(localStorage.getItem('flex_qureo_orders') || '[]');
            orders.unshift(order);
            localStorage.setItem('flex_qureo_orders', JSON.stringify(orders));
        }

        function getOrdersFromStorage() {
            return JSON.parse(localStorage.getItem('flex_qureo_orders') || '[]');
        }

        function renderAdminOrders() {

            const orders = getOrdersFromStorage();

            if(orders.length === 0) {

                    <td class="p-4 font-bold text-white">${o.studentName}</td>
                    <td class="p-4">

                        </span>
                    </td>
                    <td class="p-4 text-cyan-400 font-mono font-bold">${o.phone}</td>
                    <td class="p-4 max-w-xs truncate text-gray-300 text-xs" title="${o.details}">${o.details}</td>
                    <td class="p-4 text-center">
                        <button onclick="viewImageModal('${o.receiptImg}')" class="px-3 py-1.5 rounded-lg bg-cyan-950 hover:bg-cyan-900 border border-cyan-500/40 text-cyan-300 text-xs font-bold transition">

                            <i class="fa-solid fa-trash-can"></i>
                        </button>
                    </td>
                </tr>
            `).join('');
        }

        function deleteOrder(id) {
            let orders = getOrdersFromStorage();

            localStorage.setItem('flex_qureo_orders', JSON.stringify(orders));
            renderAdminOrders();
            showToast("تم حذف الطلب بنجاح");
        }
